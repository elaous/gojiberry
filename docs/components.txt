var FormatTime = func(t time.Time, seconds bool) string {
	if seconds {
		return t.Format("2006-01-02 15:04:05")

	return t.Format("2006-01-02 15:04")

var TranslatePlural = func(id, one, other string, n int) string {
	m := other
	if n == 1 {
		m = one
	}
	return fmt.Sprintf(m, n)
}
